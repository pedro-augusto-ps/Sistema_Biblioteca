def criar_tabela():
    conexao = sqlite3.connect("Banco_biblioteca/biblioteca.db")
    cursor = conexao.cursor()
    cursor.execute("""CREATE TABLE itens(
    id INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    autor TEXT NOT NULL,
    DISPONIBILIDADE INT NOT NULL)""")
    conexao.commit()

def criar_tabela_user():
    conexao = sqlite3.connect("Banco_biblioteca/biblioteca.db")
    cursor = conexao.cursor()
    cursor.execute("""CREATE TABLE usuarios(
    id INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL UNIQUE,
    senha TEXT NOT NULL)""")
    conexao.commit()

def criar_tabela_emprest():
    conexao = sqlite3.connect("Banco_biblioteca/biblioteca.db")
    cursor = conexao.cursor()
    cursor.execute("""CREATE TABLE emprestimos(
    id_emp INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
    usuario_ID INT NOT NULL,
    itens_ID INT NOT NULL, 
    FOREIGN KEY(usuario_ID) REFERENCES usuarios(id),
    FOREIGN KEY(itens_ID) REFERENCES itens(id))""")
    conexao.commit()
